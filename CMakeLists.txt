cmake_minimum_required(VERSION 3.18)
project(MiningPocket LANGUAGES CXX)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/benchmark/case_001.json")
 file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/benchmark/cases.zip" DESTINATION "${CMAKE_CURRENT_SOURCE_DIR}/benchmark")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(miningpocket src/main.cpp src/geometry.cpp src/evaluation.cpp src/brs.cpp src/search.cpp src/rs.cpp)
if(MSVC)
 target_compile_options(miningpocket PRIVATE /O2 /W4)
else()
 target_compile_options(miningpocket PRIVATE -O3 -Wall -Wextra)
endif()

include(CTest)
if(BUILD_TESTING)
 add_test(NAME connector_and_timing COMMAND miningpocket selftest ${CMAKE_CURRENT_SOURCE_DIR})
 add_test(NAME reference_parity COMMAND miningpocket parity ${CMAKE_CURRENT_SOURCE_DIR})
endif()
